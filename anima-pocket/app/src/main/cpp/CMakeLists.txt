cmake_minimum_required(VERSION 3.22)
project(anima_pocket LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_VULKAN ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
find_library(ANDROID_VULKAN_LIB vulkan REQUIRED)
set(Vulkan_LIBRARY "${ANDROID_VULKAN_LIB}" CACHE FILEPATH "" FORCE)
set(Vulkan_INCLUDE_DIR "${ANDROID_NDK}/toolchains/llvm/prebuilt/linux-x86_64/sysroot/usr/include" CACHE PATH "" FORCE)
find_program(Vulkan_GLSLC_EXECUTABLE glslc REQUIRED)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/stable-diffusion.cpp/CMakeLists.txt")
    message(FATAL_ERROR "Clone stable-diffusion.cpp with submodules before building.")
endif()
add_subdirectory(stable-diffusion.cpp "${CMAKE_BINARY_DIR}/sdcore")
add_library(animabridge SHARED bridge.cpp)
target_include_directories(animabridge PRIVATE stable-diffusion.cpp/include)
target_link_libraries(animabridge PRIVATE stable-diffusion log android)
